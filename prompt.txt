Sample Questions and Queries:

Question: "list all assignment groups"
Query: SELECT DISTINCT assignment_group FROM public.metrics;

Question: "Show the 5 most recent response times for WebSrvA"
Query: SELECT time, value, metadata FROM public.metrics WHERE app_name ILIKE 'websrva' AND metric_name = 'response_time' ORDER BY time DESC LIMIT 5;

Question: "get the latest status of the dynatrace synthetic on WebSrvAHost"
Query: SELECT time, status FROM public.metrics WHERE host_name ILIKE 'WebSrvAHost' AND metric_name = 'dynatrace_synth' ORDER BY time DESC LIMIT 1;

Question: "List the unique applications monitored by Nagios ownded by assignment group AetherisDevelopers"
Query: SELECT DISTINCT app_name FROM public.metrics WHERE tool_name ILIKE 'Nagios' AND assignment_group = 'AetherisDevelopers';

Question: "Show me the last 10 response times for any host in the 'prd' environment for AppID '50671'."
Query: SELECT time, host_name, value, metadata FROM public.metrics WHERE env ILIKE 'prd' AND app_id = '50671' AND metric_name = 'response_time' ORDER BY time DESC LIMIT 10;

Question: "Which hosts under the 'AetherisDevelopers' group are currently reporting a status other than 'healthy'?"
Query: SELECT DISTINCT host_name, status, metric_name  FROM public.metrics WHERE assignment_group = 'AetherisDevelopers' AND status != 'healthy'  AND time > NOW() - INTERVAL '15 minutes';

Question: "What is the average CPU load for InfraSrv1Host over the last 24 hours?"
Query: SELECT AVG(value) FROM public.metrics WHERE host_name ILIKE 'InfraSrv1Host' AND metric_name = 'cpu_load' AND time > NOW() - INTERVAL '24 hours';

Question: "What is the 5 latest values of CPU load for InfraSrv1Host?"
Query: SELECT time, value FROM public.metrics WHERE host_name ILIKE 'InfraSrv1Host' AND metric_name = 'cpu_load' ORDER BY time DESC LIMIT 5;
