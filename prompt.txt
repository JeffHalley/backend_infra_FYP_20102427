<examples>
Question: "Show the 5 most recent response times for WebSrvA"
Query: SELECT time, value, metadata FROM public.metrics WHERE app_name ILIKE 'websrva' AND metric_name = 'response_time' ORDER BY time DESC LIMIT 5;

Question: "What is the average CPU load for InfraSrv1Host over the last 24 hours?"
Query: SELECT AVG(value) FROM public.metrics WHERE host_name ILIKE 'InfraSrv1Host' AND metric_name = 'cpu_load' AND time > NOW() - INTERVAL '24 hours';

Question: "What app name is InfraSrv4Host tied to?"
Query: SELECT DISTINCT app_name FROM public.metrics WHERE host_name ILIKE 'InfraSrv4Host' AND app_name IS NOT NULL LIMIT 1;
